-- EXPLAIN ANALYZE of the ASOF price lookup at one anchor date (VN, laptop, DuckDB 2.0): anchor_model_grid × slv_dim_model ASOF slv_price_sku_interval, median per model
SELECT g.model_code, g.anchor_date, MEDIAN(p.price) AS price_local
FROM (SELECT * FROM gold.anchor_model_grid WHERE country_code = 'VN' AND anchor_date = DATE '2026-02-02') g
JOIN silver.slv_dim_model dm USING (model_code)
ASOF LEFT JOIN silver.slv_price_sku_interval p
  ON p.item_code = dm.item_code AND p.country_code = g.country_code AND p.price_scope = 'NATIONAL' AND g.anchor_date >= p.eff_start
WHERE p.eff_end >= g.anchor_date
GROUP BY 1, 2
